def build_purchase(setup, *, jewellery=500, adjustment=0, apply=0, seller=None, purchase_tax=None):
    """Booking (deposit 100 collected, TEST env) -> dispatch -> trial -> sealed purchase -> DUE."""
    graph = trial_graph(setup)
    seller = seller or setup.entity
    payable = jewellery + adjustment - apply
    purchase = Purchase.objects.create(
        booking=graph.booking,
        revision=1,
        seller=seller,
        seller_snapshot={"probe": True},
        customer_snapshot={"probe": True},
        tax_evidence=purchase_tax or graph.tax,
        currency="INR",
        jewellery_total_minor=jewellery,
        payable_minor=payable,
        balance_snapshot={"probe": True},
    )
    item = PurchaseItem.objects.create(
        purchase=purchase,
        reservation=graph.hold,
        line_number=1,
        amount_minor=jewellery,
        currency="INR",
        product_snapshot={"probe": True},
        price_snapshot={"probe": True},
    )
    basis = PurchaseBalanceEvidence.objects.create(
        purchase=purchase,
        amount_due_before_application=jewellery + adjustment,
        adjustment_minor=adjustment,
        currency="INR",
        actor=graph.actor,
        observed_at=timezone.now(),
        reason="probe",
        evidence_reference="probe",
        calculation_snapshot={"probe": True},
        policy_snapshot={"probe": True},
    )
    if apply:
        UpfrontApplication.objects.create(
            balance_evidence=basis,
            source_allocation=graph.charge.allocations.get(),
            amount_minor=apply,
            currency="INR",
            actor=graph.actor,
            observed_at=timezone.now(),
            reason="probe",
            evidence_reference="probe",
            policy_snapshot={"probe": True},
            idempotency_key=uuid.uuid4(),
        )
    purchase.status = "SEALED"
    purchase.sealed_at = timezone.now()
    purchase.save()
    obligation = PurchaseObligation.objects.create(
        purchase=purchase,
        amount_minor=payable,
        currency="INR",
        quote_expires_at=timezone.now() + timedelta(minutes=5),
    )
    obligation.status = "DUE"
    obligation.accepted_at = timezone.now()
    obligation.save()
    context = PaymentContext.objects.create(
        purchase_obligation=obligation,
        version=1,
        expires_at=timezone.now() + timedelta(minutes=3),
        policy_snapshot={"probe": True},
    )
    attempt = PaymentAttempt.objects.create(
        context=context,
        request_id=uuid.uuid4(),
        request_fingerprint="a" * 64,
        provider="FICTIONAL",
        provider_account="test-account",
        environment="TEST",
        amount_minor=payable,
        currency="INR",
    )
    graph.final = SimpleNamespace(
        purchase=purchase, item=item, obligation=obligation, context=context, attempt=attempt
    )
    return graph


def pay(graph, *, allocated_at=None):
    final = graph.final
    amount = final.obligation.amount_minor
    payment_id = f"final-{final.purchase.pk}"
    final.attempt.status = "PENDING"
    final.attempt.provider_order_id = str(uuid.uuid4())
    final.attempt.save()
    receipt(final, "AUTHORISATION", amount_minor=amount, provider_payment_id=payment_id)
    final.attempt.status = "AUTHORIZED"
    final.attempt.last_verified_at = timezone.now()
    final.attempt.save()
    final.context.status = "CAPTURING"
    final.context.capture_started_at = timezone.now()
    final.context.save()
    final.attempt.status = "CAPTURE_PENDING"
    final.attempt.capture_requested_at = timezone.now()
    final.attempt.save()
    captured = receipt(final, amount_minor=amount, provider_payment_id=payment_id)
    allocation = PaymentAllocation.objects.create(
        gateway_transaction=captured,
        purchase_obligation=final.obligation,
        amount_minor=amount,
        currency="INR",
        idempotency_key=uuid.uuid4(),
        **({"allocated_at": allocated_at} if allocated_at else {}),
    )
    final.attempt.status = "SUCCEEDED"
    final.attempt.last_verified_at = timezone.now()
    final.attempt.succeeded_at = timezone.now()
    final.attempt.save()
    final.obligation.status = "COLLECTED"
    final.obligation.save()
    final.context.status = "PAID"
    final.context.paid_at = timezone.now()
    final.context.save()
    final.purchase.status = "CONFIRMED"
    final.purchase.confirmed_at = timezone.now()
    final.purchase.save()
    return captured, allocation


def invoice(graph, *, amount, line_amount=None, number=2, issued_at=None, series=None,
            tax=None, issuer_snapshot=None, recipient_snapshot=None, line_snapshot=None,
            kind="TAX_INVOICE", corrects=None):
    document = FinancialDocument.objects.create(
        booking=graph.booking,
        purchase_obligation=graph.final.obligation,
        corrects=corrects,
        kind=kind,
        series=series or graph.series,
        issuer_snapshot=issuer_snapshot or {"x": "y"},
        recipient_snapshot=recipient_snapshot or {"x": "y"},
        tax_evidence=tax or graph.tax,
        amount_minor=amount,
        currency="INR",
        review_reference="probe-format",
    )
    FinancialDocumentLine.objects.create(
        document=document,
        line_number=1,
        amount_minor=amount if line_amount is None else line_amount,
        snapshot=line_snapshot or {"x": "y"},
        purchase_item=graph.final.item,
    )
    document.status = "ISSUED"
    document.number = number
    document.issued_at = issued_at or timezone.now()
    document.save()
    return document


def attempt_db(callable_):
    """Return True when the database/ORM rejects the action, False when it is accepted."""
    try:
        with transaction.atomic():
            callable_()
    except (DatabaseError, ValidationError) as error:
        print(f"\n    REJECTED -> {type(error).__name__}: {str(error).splitlines()[0][:160]}")
        return True
    print("\n    ACCEPTED")
    return False

@pytest.mark.parametrize("adjustment", [-100, 100])
def test_P01_adjustment_money_collected_but_never_invoiced(setup, adjustment):
    """Signed adjustment changes the amount paid; tax invoice must still equal jewellery total."""
    graph = build_purchase(setup, jewellery=500, adjustment=adjustment)
    captured, _ = pay(graph)
    assert captured.amount_minor == 500 + adjustment  # what the customer actually paid
    # An invoice for the amount actually charged is refused ...
    assert attempt_db(lambda: invoice(graph, amount=500 + adjustment, number=3))
    # ... only the pre-adjustment jewellery value can be invoiced.
    doc = invoice(graph, amount=500, number=4)
    assert doc.status == "ISSUED" and doc.amount_minor != captured.amount_minor

def test_P04_series_mixes_document_kinds_back_dates_and_exceeds_16_chars(setup):
    graph = build_purchase(setup, jewellery=500)
    pay(graph)
    # The fixture already issued DISPATCH #1 (a nil-value challan) in this same series.
    assert graph.document.kind == "DISPATCH" and graph.document.number == 1
    assert graph.document.amount_minor == 0
    long_series = DocumentSeries.objects.create(
        issuer=setup.entity,
        code="ARKATARA-BLR-TAX-INVOICE",
        scope_snapshot={"probe": True},
        review_reference="probe",
    )
    # Back-dated into the previous financial year, before the booking even existed.
    prior_fy = datetime(2026, 3, 20, 12, 0, tzinfo=dt_timezone.utc)
    assert graph.booking.created_at > prior_fy
    doc = invoice(graph, amount=500, number=98765432109876, issued_at=prior_fy, series=long_series)
    shown = f"{doc.series.code}/{doc.number}"
    assert doc.status == "ISSUED" and len(shown) > 16
    assert doc.issued_at < graph.final.purchase.confirmed_at
    # Same series as the challan also accepts a tax invoice (kinds share one number space).
    graph2_doc_ok = not attempt_db(lambda: invoice(graph, amount=500, number=7))
    assert graph2_doc_ok

def test_P06_credit_adjustments_unbounded_undirected_backdated_cross_booking(setup):
    graph = build_purchase(setup, jewellery=500)
    pay(graph)
    original = invoice(graph, amount=500, number=2)
    first = invoice(graph, amount=5000, number=3, kind="ADJUSTMENT", corrects=original,
                    issued_at=original.issued_at - timedelta(days=30))
    second = invoice(graph, amount=5000, number=4, kind="ADJUSTMENT", corrects=original)
    assert first.status == second.status == "ISSUED"
    assert first.amount_minor + second.amount_minor == 20 * original.amount_minor
    assert first.issued_at < original.issued_at  # correction dated before what it corrects
    # A correction line can point at another customer's booking stock.
    other = booking_graph(setup)
    start_payment(other, stock(setup))
    stray = FinancialDocument.objects.create(
        booking=graph.booking, purchase_obligation=graph.final.obligation, corrects=original,
        kind="ADJUSTMENT", series=graph.series, issuer_snapshot={"x": 1},
        recipient_snapshot={"x": 1}, tax_evidence=graph.tax, amount_minor=1, currency="INR",
        review_reference="probe")
    FinancialDocumentLine.objects.create(document=stray, line_number=1, amount_minor=1,
                                         snapshot={"x": 1}, reservation=other.hold)
    stray.status, stray.number, stray.issued_at = "ISSUED", 5, timezone.now()
    stray.save()
    assert other.hold.payment_context.booking_charge.booking_id != graph.booking.pk

    doc.save()
    graph.charge.refresh_from_db()
    assert doc.status == "ISSUED" and graph.charge.status == "QUOTED"

def test_P08_collected_upfront_funds_can_never_be_discharged(setup):
    graph = trial_graph(setup)  # deposit 100 collected, booking confirmed
    names = {model.__name__.lower() for model in apps.get_models()}
    assert not {n for n in names if "refund" in n or "settlement" in n or "retention" in n}
    # Charge is terminal at COLLECTED and the paid context is terminal: no void/refund/retain path.
    assert attempt_db(lambda: (setattr(graph.charge, "status", "VOID"), graph.charge.save()))
    graph.charge.refresh_from_db()
    assert attempt_db(
        lambda: connection.cursor().execute(
            "UPDATE payments_bookingcharge SET status='VOID' WHERE id=%s", [graph.charge.pk]
        )
    )
    # Gateway evidence cannot record a refund or chargeback at all.
    assert attempt_db(
        lambda: raw_insert(
            GatewayTransaction, attempt=graph.attempt, provider="FICTIONAL",
            provider_account="test-account", environment="TEST", provider_payment_id="rfnd",
            transaction_kind="REFUND", amount_minor=100, currency="INR",
            occurred_at=timezone.now(), verified_at=timezone.now(), verification_reference="x")
    )
    unapplied = graph.charge.allocations.get().amount_minor - sum(
        a.amount_minor for a in UpfrontApplication.objects.filter(
            source_allocation__booking_charge=graph.charge))
    assert unapplied == 100  # remains an open liability with no possible closing record
